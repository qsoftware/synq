cmake_minimum_required(VERSION 3.14)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

project(synq VERSION 0.0.1)


include(FetchContent)

# --------------------
# build options
# --------------------
# When built through pip (scikit-build-core sets SKBUILD) the Python module is
# built and the C++ tests are skipped, so GTest is not required.
if (DEFINED SKBUILD)
    set(_synq_default_python ON)
    set(_synq_default_tests OFF)
else()
    set(_synq_default_python OFF)
    set(_synq_default_tests ON)
endif()
option(SYNQ_BUILD_TESTS "Build the C++ test suite (requires GTest)" ${_synq_default_tests})
option(SYNQ_BUILD_PYTHON "Build the Python module synq._synq (requires pybind11)" ${_synq_default_python})

# --------------------
# google test
# --------------------
if (SYNQ_BUILD_TESTS)
    find_package(GTest CONFIG REQUIRED)
endif()


# --------------------
# Eigen
# --------------------
find_package(Eigen3 REQUIRED NO_MODULE)

# --------------------
# MKL(WIN32) or Lapack(UNIX)
# --------------------
if (WIN32)
    add_definitions(-DUSE_MKL)
    find_package(MKL CONFIG REQUIRED)
else()
    find_package(LAPACK REQUIRED)
    find_library(LAPACKE_LIBRARY lapacke REQUIRED)
    find_path(LAPACKE_INCLUDE_DIR lapacke.h REQUIRED)
endif()

add_library(synq

        src/iast.cpp

        src/qsd.cpp

# The static library is linked into the Python extension, so it must be PIC.
set_target_properties(synq PROPERTIES POSITION_INDEPENDENT_CODE ON)


if(WIN32)
    target_link_libraries(synq PUBLIC
        Eigen3::Eigen
        MKL::MKL)
else()
    target_include_directories(synq PUBLIC ${LAPACKE_INCLUDE_DIR})
    target_link_libraries(synq PUBLIC
        Eigen3::Eigen
        ${LAPACKE_LIBRARY}
        ${LAPACK_LIBRARIES}
        ${BLAS_LIBRARIES})
endif()

# --------------------
# Python module (synq._synq)
# --------------------
if (SYNQ_BUILD_PYTHON)
    find_package(Python 3.8 REQUIRED COMPONENTS Interpreter Development.Module)
    find_package(pybind11 CONFIG REQUIRED)

    pybind11_add_module(_synq python/bindings.cpp)
    target_link_libraries(_synq PRIVATE synq)
    target_compile_definitions(_synq PRIVATE SYNQ_VERSION="${PROJECT_VERSION}")

    # Directories with runtime DLLs (Windows). Python >= 3.8 ignores PATH when
    # loading extension dependencies, so synq/__init__.py registers these with
    # os.add_dll_directory. MKL also loads some of its DLLs dynamically
    # (mkl_avx2, ...), so copying the DLLs next to the module is not enough.
    set(_synq_dll_dirs "")
    if (WIN32)
        if (DEFINED VCPKG_INSTALLED_DIR AND DEFINED VCPKG_TARGET_TRIPLET)
            list(APPEND _synq_dll_dirs "${VCPKG_INSTALLED_DIR}/${VCPKG_TARGET_TRIPLET}/bin")
        endif()
        if (DEFINED MKL_DIR)
            get_filename_component(_mkl_bin "${MKL_DIR}/../../bin" ABSOLUTE)
            list(APPEND _synq_dll_dirs "${_mkl_bin}")
        endif()
        list(REMOVE_DUPLICATES _synq_dll_dirs)
    endif()
    set(_synq_dll_dirs_py "")
    foreach (_dir IN LISTS _synq_dll_dirs)
        string(REPLACE "\\" "/" _dir "${_dir}")
        string(APPEND _synq_dll_dirs_py "    \"${_dir}\",\n")
    endforeach()
    file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/_build_config.py"
         "\"\"\"Generated by CMake at build time.\"\"\"\n\nDLL_DIRS = [\n${_synq_dll_dirs_py}]\n")

    if (DEFINED SKBUILD)
        install(TARGETS _synq LIBRARY DESTINATION synq RUNTIME DESTINATION synq)
        install(FILES "${CMAKE_CURRENT_BINARY_DIR}/_build_config.py" DESTINATION synq)
    endif()
endif()
